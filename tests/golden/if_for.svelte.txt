<script>
  let { title = 'Items' } = $props();
  let showHint = $state(1);
  function setShowHint(v) { showHint = typeof v === 'function' ? v(showHint) : v; }
</script>

<div class="flex flex-col gap-16 p-24">
  <h1 class="text-2xl font-bold">
    {title}
  </h1>
  {#if showHint}
    <p class="text-muted">
      Hint: click items
    </p>
  {/if}
  {#each items || [] as item (item.id)}
    <div class="flex flex-row gap-8 flex items-center justify-center">
      <span class="font-bold">
        {item.name}
      </span>
      <button class="btn btn-outline" onclick={() => selectItem(item.id)}>
        Select
      </button>
    </div>
  {/each}
</div>
