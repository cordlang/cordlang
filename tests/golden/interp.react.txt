import { useState } from 'react';

export default function Greeter({ userName = "friend" }) {
  const [visits, setVisits] = useState(1);
  return (
    <>
      <div className="flex flex-col gap-12 p-24">
        <h1 className="text-2xl font-bold">
          {'Hola, '}
          {userName}
          {'!'}
        </h1>
        <p className="text-muted">
          {'Visitas: '}
          {visits}
        </p>
        <p className="text-lg">
          {'Total: $'}
          {visits}
        </p>
        <button className="btn btn-primary" onClick={() => setVisits(visits + 1)}>
          {'Cart ('}
          {visits}
          {')'}
        </button>
        <button className="btn btn-outline" onClick={() => setVisits(0)}>
          {'Reset'}
        </button>
      </div>
    </>
  );
}

// ===== src/App.jsx =====
/* cordlang: source=src/app.cord */
import Greeter from './components/Greeter';

export default function App() {
  return <Greeter />;
}
