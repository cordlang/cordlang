<script>
  let { userName = 'friend' } = $props();
  let visits = $state(1);
  function setVisits(v) { visits = typeof v === 'function' ? v(visits) : v; }
</script>

<div class="flex flex-col gap-12 p-24">
  <h1 class="text-2xl font-bold">
    Hola, 
    {userName}
    !
  </h1>
  <p class="text-muted">
    Visitas: 
    {visits}
  </p>
  <p class="text-lg">
    Total: $
    {visits}
  </p>
  <button class="btn btn-primary" onclick={() => setVisits(visits + 1)}>
    Cart (
    {visits}
    )
  </button>
  <button class="btn btn-outline" onclick={() => setVisits(0)}>
    Reset
  </button>
</div>
