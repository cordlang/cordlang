<!-- ===== src/router.js ===== -->
import { createRouter, createWebHashHistory } from 'vue-router';
import DefaultLayout from './layouts/DefaultLayout.vue';
import Shop from './layouts/Shop.vue';
import HomePage from './pages/HomePage.vue';
import ShopPage from './pages/ShopPage.vue';
import ProductPage from './pages/ProductPage.vue';

const routes = [
  { path: '/', component: HomePage, meta: { layout: DefaultLayout } },
  { path: '/shop', component: ShopPage, meta: { layout: Shop } },
  { path: '/shop/:id', component: ProductPage, meta: { layout: Shop } },
];

export const router = createRouter({
  history: createWebHashHistory(),
  routes,
});

<!-- ===== src/App.vue ===== -->
<!-- cordlang: source=src/app.cord -->
<script setup>
import { RouterView } from 'vue-router';
import { computed } from 'vue';
import { router } from './router.js';
const layout = computed(() => router?.currentRoute.value.meta.layout || null);
</script>

<template>
  <component :is="layout" v-if="layout">
    <RouterView />
  </component>
  <RouterView v-else />
</template>
