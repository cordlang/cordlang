<script setup>
import { ref } from 'vue';
const props = defineProps({
  label: { default: 'Counter' },
});
const label = props.label;
const count = ref(0);
function setCount(v) { count.value = typeof v === 'function' ? v(count.value) : v; }
</script>

<template>
<div class="flex flex-col gap-16 p-24 flex items-center justify-center">
  <h1 class="text-2xl font-bold">
    {{ label }}
  </h1>
  <p class="text-lg text-muted">
    Count: 
    {{ count }}
  </p>
  <div class="flex flex-row gap-8 flex items-center justify-center">
    <button class="btn btn-outline" @click="() => setCount(count - 1)">
      -
    </button>
    <span class="text-4xl font-bold">
      {{ count }}
    </span>
    <button class="btn btn-primary" @click="() => setCount(count + 1)">
      +
    </button>
  </div>
  <button class="btn btn-ghost" @click="() => setCount(0)">
    Reset
  </button>
</div>
</template>

export const router = null;
