<!-- ===== src/App.svelte ===== -->
<!-- cordlang: source=src/app.cord -->
<script>
  import { onMount } from 'svelte';
  import DefaultLayout from './layouts/DefaultLayout.svelte';
  import Shop from './layouts/Shop.svelte';
  import HomePage from './pages/HomePage.svelte';
  import ShopPage from './pages/ShopPage.svelte';
  import ProductPage from './pages/ProductPage.svelte';

  const routes = [
    { path: '/', component: HomePage, layout: DefaultLayout },
    { path: '/shop', component: ShopPage, layout: Shop },
    { path: '/shop/:id', component: ProductPage, layout: Shop },
  ];

  function currentPath() {
    const raw = window.location.hash.replace(/^#/, '') || '/';
    const h = raw.split('?')[0] || '/';
    return h.startsWith('/') ? h : '/' + h;
  }
  function matchRoute(pathname) {
    const segs = pathname.replace(/\/+$/, '').split('/').filter(Boolean);
    for (const r of routes) {
      const pp = r.path.replace(/\/+$/, '').split('/').filter(Boolean);
      if (pp.length !== segs.length) continue;
      const params = {};
      let ok = true;
      for (let i = 0; i < pp.length; i++) {
        if (pp[i].startsWith(':')) params[pp[i].slice(1)] = decodeURIComponent(segs[i]);
        else if (pp[i] !== segs[i]) { ok = false; break; }
      }
      if (ok) return { component: r.component, layout: r.layout, params };
    }
    return null;
  }
  let path = $state(currentPath());
  onMount(() => {
    const onHash = () => { path = currentPath(); };
    window.addEventListener('hashchange', onHash);
    if (!window.location.hash) window.location.hash = '#/';
    return () => window.removeEventListener('hashchange', onHash);
  });
  let matched = $derived(matchRoute(path) ?? matchRoute('/') ?? null);
  let Page = $derived(matched?.component ?? null);
  let Layout = $derived(matched?.layout ?? null);
  let params = $derived(matched?.params ?? {});
</script>

{#if Layout && Page}
  <Layout>
    {#snippet children()}
      <Page {...params} />
    {/snippet}
  </Layout>
{:else if Page}
  <Page {...params} />
{/if}
