import { createSignal } from 'solid-js';

export default function Greeter({ userName = "friend" }) {
  const [visits, setVisits] = createSignal(1);
  return (
    <>
      <div class="flex flex-col gap-12 p-24">
        <h1 class="text-2xl font-bold">
          {'Hola, '}
          {userName}
          {'!'}
        </h1>
        <p class="text-muted">
          {'Visitas: '}
          {visits()}
        </p>
        <p class="text-lg">
          {'Total: $'}
          {visits()}
        </p>
        <button class="btn btn-primary" onClick={() => setVisits(visits() + 1)}>
          {'Cart ('}
          {visits()}
          {')'}
        </button>
        <button class="btn btn-outline" onClick={() => setVisits(0)}>
          {'Reset'}
        </button>
      </div>
    </>
  );
}

// ===== src/App.jsx =====
/* cordlang: source=src/app.cord */
import Greeter from './components/Greeter';

export default function App() {
  return <Greeter />;
}
