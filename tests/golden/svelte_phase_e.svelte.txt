<script>
  import { writable } from 'svelte/store';
  import { fade } from 'svelte/transition';
  const n = writable(0);
  function setN(v) { n.set(v); }
  function portal(node, target) {
    const t = target || (typeof document !== 'undefined' ? document.body : null);
    if (t) t.appendChild(node);
    return {
      update(newTarget) {
        const nt = newTarget || (typeof document !== 'undefined' ? document.body : null);
        if (nt) nt.appendChild(node);
      },
      destroy() { node.remove(); }
    };
  }
</script>

{#snippet box(label)}
  <p>
    {label}
  </p>
{/snippet}
