<script setup>
import { ref } from 'vue';
const props = defineProps({
  userName: { default: 'friend' },
});
const userName = props.userName;
const visits = ref(1);
function setVisits(v) { visits.value = typeof v === 'function' ? v(visits.value) : v; }
</script>

<template>
<div class="flex flex-col gap-12 p-24">
  <h1 class="text-2xl font-bold">
    Hola, 
    {{ userName }}
    !
  </h1>
  <p class="text-muted">
    Visitas: 
    {{ visits }}
  </p>
  <p class="text-lg">
    Total: $
    {{ visits }}
  </p>
  <button class="btn btn-primary" @click="() => setVisits(visits + 1)">
    Cart (
    {{ visits }}
    )
  </button>
  <button class="btn btn-outline" @click="() => setVisits(0)">
    Reset
  </button>
</div>
</template>

export const router = null;
